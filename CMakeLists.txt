cmake_minimum_required(VERSION 3.16)
project(greenplan_parallel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(greenplan src/main.cpp)
find_package(OpenMP QUIET)
if(OpenMP_CXX_FOUND)
  target_link_libraries(greenplan PRIVATE OpenMP::OpenMP_CXX)
endif()
option(ENABLE_MPI "Build MPI evaluation backend" OFF)
if(ENABLE_MPI)
  find_package(MPI REQUIRED)
  target_link_libraries(greenplan PRIVATE MPI::MPI_CXX)
  target_compile_definitions(greenplan PRIVATE HAVE_MPI)
endif()
if(MSVC)
  target_compile_options(greenplan PRIVATE /W4 /O2 /utf-8)
else()
  target_compile_options(greenplan PRIVATE -Wall -Wextra -Wpedantic -O2)
endif()
enable_testing()
add_executable(model_test tests/model_test.cpp)
if(MSVC)
  target_compile_options(model_test PRIVATE /UNDEBUG)
else()
  target_compile_options(model_test PRIVATE -UNDEBUG)
endif()
add_test(NAME model_invariants COMMAND model_test)
add_test(NAME smoke COMMAND greenplan --population 20 --generations 3 --months 2 --output smoke.csv)
